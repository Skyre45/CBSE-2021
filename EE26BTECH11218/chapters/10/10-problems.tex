\item $PQ$ is a tangent to a circle with centre $O$ at the point $P$ on the circle. If $\triangle OPQ$ is an isosceles triangle, then find $\angle OQP$.
\hfill (10,2021)
\\
\solution
From \eqref{eq:EE26BTECH11218-cb-tan-radius}, the direction $\vec{m}=\vec{Q}-\vec{P}$ of the tangent satisfies
\begin{align}
\brak{\vec{Q}-\vec{P}}^{\top}\brak{\vec{P}-\vec{O}} = 0 \implies \angle OPQ = 90^{\circ}
\end{align}
Hence, by Pythagoras,
\begin{align}
\norm{\vec{O}-\vec{Q}}^2 = \norm{\vec{O}-\vec{P}}^2 + \norm{\vec{P}-\vec{Q}}^2
\end{align}
so $OQ$ is strictly the longest side, and the equal sides of the isosceles triangle must be $OP$ and $PQ$. Since the shape is fixed up to rotation and scaling, take
\begin{align}
\vec{O} = \myvec{0 \\ 0}, \quad \vec{P} = \myvec{r \\ 0}, \quad \vec{m} = \myvec{0 \\ 1}, \quad \vec{Q} = \vec{P} + r\vec{m} = \myvec{r \\ r}
\end{align}
Substituting in \eqref{eq:EE26BTECH11218-cb-angle},
\begin{align}
\cos\angle OQP = \frac{\brak{\vec{O}-\vec{Q}}^{\top}\brak{\vec{P}-\vec{Q}}}{\norm{\vec{O}-\vec{Q}}\norm{\vec{P}-\vec{Q}}}
= \frac{\myvec{-r & -r}\myvec{0 \\ -r}}{r\sqrt{2}\cdot r} = \frac{1}{\sqrt{2}}
\end{align}
\begin{align}
\therefore \angle OQP = 45^{\circ}
\end{align}

\hfill (10,2021)

\begin{enumerate}[label=\alph*), itemsep=0pt]

        \item $\frac{1}{10}$

        \item $\frac{1}{10}$
        \item $0$
        \item $\frac{1}{5}$
    \end{multicols}
\end{enumerate}

\solution
Let $X$ be the number on the card drawn. It takes values in $\cbrak{1,2,\dots,20}$ with the PMF $p_X\brak{n}=\pr{X=n}=\frac{1}{20}$. For an event $E$, define the indicator random variable $Y_E$ by $Y_E=1$ if $X\in E$ and $Y_E=0$ otherwise. From \eqref{eq:EE26BTECH11218-cb-prob},
\begin{align}
\pr{E} = \pr{Y_E=1} = \sum_{n\in E}p_X\brak{n} = \frac{\abs{E}}{20}
\end{align}
The primes up to $20$ are $2,3,5,7,11,13,17,19$, so there are $8$ of them.

\item $ABC$ is a right triangle, right-angled at $B$, with $BC=6$ cm and $AB=8$ cm. A circle with centre $O$ and radius $r$ cm is inscribed in $\triangle ABC$. Find the value of $r$.
\hfill (10,2021)
\\
\solution

\begin{align}
\vec{A} = \myvec{0 \\ 8}, \quad \vec{B} = \myvec{0 \\ 0}, \quad \vec{C} = \myvec{6 \\ 0}
\end{align}
The three sides of the triangle are $BC$, $CA$ and $AB$ (the side $AB$ joins $A$ and $B$, and so on), with lengths $a=BC=6$, $b=CA=\norm{\vec{C}-\vec{A}}=10$ and $c=AB=8$. The centre $O$ of the inscribed circle is the incentre $\vec{I}$, so from \eqref{eq:EE26BTECH11218-cb-incircle},

r = \frac{\mathcal{A}}{s} = \frac{\frac{1}{2}\cdot6\cdot8}{12} = 2
\end{align}
and from \eqref{eq:EE26BTECH11218-cb-incircle-right}, $r = \frac{6+8-10}{2} = 2$.

\textbf{A second method, from the definition of the incircle.} The circle touches all three sides, so its centre $\vec{I}=\myvec{x \\ y}$ is at the same distance $r$ from the three lines that contain the sides:
\begin{align}
\text{line } AB: \ x=0, \qquad \text{line } BC: \ y=0, \qquad \text{line } CA: \ 4x+3y=24
\end{align}
(the line $CA$ passes through $C\brak{6,0}$ and $A\brak{0,8}$). The centre lies inside the triangle, where $x>0$, $y>0$ and $4x+3y<24$, so its distances from the three lines are

\text{from } CA: \ \frac{24-4x-3y}{5} = 4.8-0.8x-0.6y
\end{align}

\begin{align}
x - r = 0, \qquad y - r = 0, \qquad -0.8x - 0.6y - r = -4.8

\item Divide the polynomial $f\brak{x} = 5x^3+10x^2-30x-15$ by the polynomial $g\brak{x} = x^2+1+x$ and hence find the quotient and the remainder.
\hfill (10,2021)

\solution

\begin{align}
q\brak{x}g\brak{x} + r\brak{x} = q_1x^3 + \brak{q_1+q_0}x^2 + \brak{q_1+q_0+r_1}x + \brak{q_0+r_0}
\end{align}
Comparing the coefficients of $x^3$, $x^2$, $x$ and $1$ with those of $f\brak{x}$ gives one equation per row of the matrix equation, in the unknown coefficients of $q$ and $r$
\begin{align}
\myvec{1 & 0 & 0 & 0 \\ 1 & 1 & 0 & 0 \\ 1 & 1 & 1 & 0 \\ 0 & 1 & 0 & 1}\myvec{q_1 \\ q_0 \\ r_1 \\ r_0} = \myvec{5 \\ 10 \\ -30 \\ -15}

\end{align}

\begin{align}
f\brak{\omega} = 5 + 10\omega^2 - 30\omega - 15 = -20-40\omega = r\brak{\omega}
\end{align}
and similarly for $\omega^2$, which gives $r\brak{x}=-40x-20$.
\begin{align}
\therefore q\brak{x} = 5x+5, \quad r\brak{x} = -40x-20

\item A man on the top of a vertical tower observes a car moving at a uniform speed coming directly towards it. If it takes $18$ minutes for the angle of depression to change from $30^{\circ}$ to $60^{\circ}$, how soon after this will the car reach the tower?
\hfill (10,2021)
\\
\solution
Let the tower have height $h$ and let $x\brak{t}$ be the horizontal distance of the car from its foot, with $t=0$ at the first observation. From \eqref{eq:EE26BTECH11218-cb-depression},
\begin{align}
x\brak{0} = h\cot30^{\circ} = \sqrt{3}h, \qquad x\brak{18} = h\cot60^{\circ} = \frac{h}{\sqrt{3}}
\end{align}
Using \eqref{eq:EE26BTECH11218-cb-uniform} and \eqref{eq:EE26BTECH11218-cb-uniform-sys} with $t_1=0$, $t_2=18$. The first row says $x_0=x\brak{0}$ and the second says $x_0-18v=x\brak{18}$, so the unknowns are the starting distance $x_0$ and the speed $v$ (distance per minute):
\begin{align}
\myvec{1 & 0 \\ 1 & -18}\myvec{x_0 \\ v} = \myvec{\sqrt{3}h \\ \frac{h}{\sqrt{3}}}
\end{align}
Gauss elimination with $R_2\to R_2-R_1$ gives
\begin{align}
\augvec{2}{1}{1 & 0 & \sqrt{3}h \\ 1 & -18 & \frac{h}{\sqrt{3}}} &\longrightarrow \augvec{2}{1}{1 & 0 & \sqrt{3}h \\ 0 & -18 & -\frac{2h}{\sqrt{3}}} \\
&\implies x_0 = \sqrt{3}h, \quad v = \frac{2h}{18\sqrt{3}} = \frac{h}{9\sqrt{3}}
\end{align}
The car reaches the tower when $x\brak{t}=0$, that is after a further time
\begin{align}
t^{*} = \frac{x\brak{18}}{v} = \frac{h/\sqrt{3}}{h/\brak{9\sqrt{3}}} = 9 \text{ minutes}
\end{align}
which is independent of $h$. The following code verifies this in \figref{fig:EE26BTECH11218-cb_q35}.
